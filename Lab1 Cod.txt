int stringtonumber(char *a)
{
    int nr = 0;
    int i = 0;
    int semn = 1;
    if(a[i] == '-')
    { 
       semn = -1;
       i++;
    }
    while(a[i]!='\0')
    {
        nr = nr * 10 + a[i] - '0';
        i++;
    }
    return nr * semn;
}
int main()
{
    char nr[300];
    while(fscanf(fp,"%s",nr)==1)
    {
        s += stringtonumber(nr);
    }
    printf("%d \n",s);
    return 0;
}

#include <iostream>
using namespace std;

bool isPrime(int n)
{
    for (int tr = 2; tr < n / 2; tr++)
        if ((n % tr ) == 0)
            return false;
    return true;
}
int main()
{
    int n;
    std::cout << "Enter a number:";
    std::cin >> n;
    if (isPrime(n))
        std::cout << n << " is prime !";
    else 
        std::cout << n << " is NOT prime !";
    return 0;
}
